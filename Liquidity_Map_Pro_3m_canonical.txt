f_samePrice(float a, float b) =>
    not na(a) and
     not na(b) and
     math.abs(a - b) <= syminfo.mintick * 0.5

f_keepNearest(
    array<int> idxArr,
    array<float> distArr,
    int idx,
    float dist,
    int limit) =>

    int insertAt = array.size(distArr)

    if array.size(distArr) > 0
        for k = 0 to array.size(distArr) - 1
            if dist < array.get(distArr, k)
                insertAt := k
                break

    array.insert(idxArr, insertAt, idx)
    array.insert(distArr, insertAt, dist)

    if array.size(idxArr) > limit
        array.pop(idxArr)
        array.pop(distArr)

    true

f_hasLevelAtPrice(float price, bool isHigh) =>
    bool found = false

    if array.size(levelPrice) > 0
        for i = 0 to array.size(levelPrice) - 1
            if array.get(levelHigh, i) == isHigh and f_samePrice(array.get(levelPrice, i), price)
                found := true
                break

    found

f_nyPreviousDayState() =>
    var float currentDayHigh = na
    var float currentDayLow = na
    var int currentDayHighTime = na
    var int currentDayLowTime = na

    var float pdh = na
    var float pdl = na
    var int pdhTime = na
    var int pdlTime = na

    var bool pdhSwept = false
    var bool pdlSwept = false

    var int pdhSweepTime = na
    var int pdlSweepTime = na

    int nyYear = year(time, "America/New_York")
    int nyMonth = month(time, "America/New_York")
    int nyDay = dayofmonth(time, "America/New_York")

    bool newNyDay = nyDay != nyDay[1] or nyMonth != nyMonth[1] or nyYear != nyYear[1]

    if na(currentDayHigh)
        currentDayHigh := high
        currentDayLow := low
        currentDayHighTime := time
        currentDayLowTime := time

    else
        if newNyDay
            pdh := currentDayHigh
            pdl := currentDayLow

            pdhTime := currentDayHighTime
            pdlTime := currentDayLowTime

            pdhSwept := false
            pdlSwept := false

            pdhSweepTime := na
            pdlSweepTime := na

            currentDayHigh := high
            currentDayLow := low
            currentDayHighTime := time
            currentDayLowTime := time

        else
            if high > currentDayHigh
                currentDayHigh := high
                currentDayHighTime := time

            if low < currentDayLow
                currentDayLow := low
                currentDayLowTime := time

    if not na(pdh) and not pdhSwept
        bool hitPDH = sweepByWick ? high >= pdh + syminfo.mintick : close >= pdh + syminfo.mintick

        if hitPDH
            pdhSwept := true
            pdhSweepTime := time

    if not na(pdl) and not pdlSwept
        bool hitPDL = sweepByWick ? low <= pdl - syminfo.mintick : close <= pdl - syminfo.mintick

        if hitPDL
            pdlSwept := true
            pdlSweepTime := time

    [
        pdh,
        pdl,
        pdhTime,
        pdlTime,
        pdhSwept,
        pdlSwept,
        pdhSweepTime,
        pdlSweepTime
    ]

    bool inPm = nyMinutes >= 13 * 60 and nyMinutes < 17 * 60

    bool pmEnds = not inPm and inPm[1]

        pmHigh := high

        pmHighSwept := false
        pmLowSwept := false

        pmHighSweepTime := na

        if high > pmHigh
            pmHigh := high
            pmHighTime := time

        if low < pmLow
            pmLow := low
            pmLowTime := time

    if not inPm and not na(pmHigh) and not pmHighSwept

        if hitLow
            pmLowSwept := true

        pmLow,
        pmHighTime,
        pmLowTime,

f_addOrUpgradeLevel(
     float price,
     bool isHigh,
     string source,
     color levelColor,
     int priority,
     int startTime) =>

    int existing = -1

    bool alreadySweptSameLevel = false

    if not na(price) and not na(startTime)

        if array.size(levelPrice) > 0
            for i = 0 to array.size(levelPrice) - 1

                bool sameSide =
                     array.get(levelHigh, i) == isHigh

                bool samePrice =
                     f_samePrice(
                         array.get(levelPrice, i),
                         price)

                bool isActive = array.get(levelActive, i)
                line existingLine = array.get(levelLine, i)
                bool sameStart = line.get_x1(existingLine) == startTime

                if sameSide and samePrice
                    if isActive
                        existing := i
                        break
                    else if sameStart
                        alreadySweptSameLevel := true
                        break


        // EXISTIERENDES LEVEL
        if existing >= 0

            int oldPriority =
                 array.get(levelPriority, existing)

            array.set(levelWasSwing, existing, array.get(levelWasSwing, existing) or priority <= 2)

            if priority > oldPriority

                array.set(
                     levelPrice,
                     existing,
                     price)

                array.set(
                     levelSource,
                     existing,
                     source)

                array.set(
                     levelPriority,
                     existing,
                     priority)

                line ln =
                     array.get(levelLine, existing)

                label lbl =
                     array.get(levelLabel, existing)

                line.set_x1(
                     ln,
                     startTime)

                line.set_y1(
                     ln,
                     price)

                line.set_x2(
                     ln,
                     time)

                line.set_y2(
                     ln,
                     price)

                line.set_extend(
                     ln,
                     extend.right)

                line.set_style(
                     ln,
                     line.style_solid)

                line.set_color(
                     ln,
                     levelColor)

                line.set_width(
                     ln,
                     priority >= 4 ? 2 : 1)

                label.set_y(
                     lbl,
                     price)

                label.set_text(
                     lbl,
                     source)

                label.set_textcolor(
                     lbl,
                     levelColor)

                label.set_style(
                     lbl,
                     label.style_none)

                label.set_size(
                     lbl,
                     labelSizeValue)


        // NEUES LEVEL
        else if not alreadySweptSameLevel

            line newLine = line.new(
                 x1=startTime,
                 y1=price,
                 x2=time,
                 y2=price,
                 xloc=xloc.bar_time,
                 extend=extend.right,
                 color=levelColor,
                 width=priority >= 4 ? 2 : 1)

            label newLabel = label.new(
                 x=startTime,
                 y=price,
                 text=source,
                 xloc=xloc.bar_time,
                 style=label.style_none,
                 textcolor=levelColor,
                 color=color.new(color.black, 100),
                 size=labelSizeValue)

            array.push(
                 levelPrice,
                 price)

            array.push(
                 levelHigh,
                 isHigh)

            array.push(
                 levelActive,
                 true)

            array.push(
                 levelSource,
                 source)

            array.push(
                 levelPriority,
                 priority)

            array.push(levelWasSwing, priority <= 2)

            array.push(
                 levelLine,
                 newLine)

            array.push(
                 levelLabel,
                 newLabel)

            array.push(levelCreatedTime, time)


            // Speicher bereinigen:
            // zuerst ein bereits gesweeptes/inaktives Level entfernen
            if array.size(levelPrice) > maxLevels

                int removeIndex = -1

                for r = 0 to array.size(levelPrice) - 1
                    if not array.get(levelActive, r)
                        removeIndex := r
                        break

                if removeIndex >= 0

                    line oldLine = array.get(levelLine, removeIndex)
                    label oldLabel = array.get(levelLabel, removeIndex)

                    line.delete(oldLine)
                    label.delete(oldLabel)

                    array.remove(levelPrice, removeIndex)
                    array.remove(levelHigh, removeIndex)
                    array.remove(levelActive, removeIndex)
                    array.remove(levelSource, removeIndex)
                    array.remove(levelPriority, removeIndex)
                    array.remove(levelWasSwing, removeIndex)
                    array.remove(levelLine, removeIndex)
                    array.remove(levelLabel, removeIndex)
                    array.remove(levelCreatedTime, removeIndex)

        true

// HIGH:
// Variante 1:
// links tiefer → Swing → rechts tiefer
//
// Variante 2:
// links tiefer → 2 gleiche Highs → rechts tiefer

f_h1HighData() =>

    // Nur abgeschlossene H1-Kerzen verwenden.
    // [1] = vollständig geschlossene rechte Bestätigungskerze
    // [2] = Swing-Kerze bei einfachem Swing

    bool singleHigh =
         high[2] > high[3] and
         high[2] > high[1]

    // Doppel-Swing:
    // links tiefer → zwei gleiche Highs → rechts tiefer
    bool doubleHigh =
         high[3] > high[4] and
         high[3] == high[2] and
         high[2] > high[1]

    bool isSwing =
         singleHigh or doubleHigh

    float swingPrice =
         isSwing ? high[2] : na

    int swingTime =
         doubleHigh ?
         time[3] :
         singleHigh ?
         time[2] :
         na

    float comparisonHigh =
         doubleHigh ?
         math.max(high[4], high[1]) :
         math.max(high[3], high[1])

    bool isSharp =
         isSwing and
         swingPrice - comparisonHigh >= sharpDistance

    [
        isSwing,
        swingPrice,
        swingTime,
        isSharp
    ]


// LOW:
// Variante 1:
// links höher → Swing → rechts höher
//
// Variante 2:
// links höher → 2 gleiche Lows → rechts höher

f_h1LowData() =>

    // Nur abgeschlossene H1-Kerzen verwenden.

    bool singleLow =
         low[2] < low[3] and
         low[2] < low[1]

    // Doppel-Swing:
    // links höher → zwei gleiche Lows → rechts höher
    bool doubleLow =
         low[3] < low[4] and
         low[3] == low[2] and
         low[2] < low[1]

    bool isSwing =
         singleLow or doubleLow

    float swingPrice =
         isSwing ? low[2] : na

    int swingTime =
         doubleLow ?
         time[3] :
         singleLow ?
         time[2] :
         na

    float comparisonLow =
         doubleLow ?
         math.min(low[4], low[1]) :
         math.min(low[3], low[1])

    bool isSharp =
         isSwing and
         comparisonLow - swingPrice >= sharpDistance

    [
        isSwing,
        swingPrice,
        swingTime,
        isSharp
    ]

f_h1HistoricalLowSnapshot() =>
    var array<float> histLowPrice = array.new_float()
    var array<int> histLowTime = array.new_int()
    var array<bool> histLowSharp = array.new_bool()

    // Bereits gespeicherte H1-Lows auf Sweep prüfen.
    int k = array.size(histLowPrice) - 1

    while k >= 0
        float storedPrice = array.get(histLowPrice, k)

        bool broken = sweepByWick ?
             low <= storedPrice - syminfo.mintick :
             close <= storedPrice - syminfo.mintick

        if broken
            array.remove(histLowPrice, k)
            array.remove(histLowTime, k)
            array.remove(histLowSharp, k)

        k -= 1

    // Neuen bestätigten H1-Swing erkennen.
    [lowSwing, lowPrice, lowTime, lowSharp] = f_h1LowData()

    if lowSwing and not na(lowPrice) and not na(lowTime)

        bool alreadyStored = false

        if array.size(histLowTime) > 0
            for n = 0 to array.size(histLowTime) - 1
                if array.get(histLowTime, n) == lowTime
                    alreadyStored := true
                    break

        if not alreadyStored
            array.push(histLowPrice, lowPrice)
            array.push(histLowTime, lowTime)
            array.push(histLowSharp, lowSharp)

    // Die drei nächsten noch offenen Lows bestimmen.
    float p1 = na
    float p2 = na
    float p3 = na

    int t1 = na
    int t2 = na
    int t3 = na

    bool s1 = false
    bool s2 = false
    bool s3 = false

    float d1 = na
    float d2 = na
    float d3 = na

    if array.size(histLowPrice) > 0
        for n = 0 to array.size(histLowPrice) - 1

            float p = array.get(histLowPrice, n)

            if p < close
                float d = close - p
                int t = array.get(histLowTime, n)
                bool s = array.get(histLowSharp, n)

                if na(d1) or d < d1
                    d3 := d2
                    p3 := p2
                    t3 := t2
                    s3 := s2

                    d2 := d1
                    p2 := p1
                    t2 := t1
                    s2 := s1

                    d1 := d
                    p1 := p
                    t1 := t
                    s1 := s

                else if na(d2) or d < d2
                    d3 := d2
                    p3 := p2
                    t3 := t2
                    s3 := s2

                    d2 := d
                    p2 := p
                    t2 := t
                    s2 := s

                else if na(d3) or d < d3
                    d3 := d
                    p3 := p
                    t3 := t
                    s3 := s

    [p1, t1, s1, p2, t2, s2, p3, t3, s3]

f_h1HistoricalHighSnapshot() =>
    var array<float> histHighPrice = array.new_float()
    var array<int> histHighTime = array.new_int()
    var array<bool> histHighSharp = array.new_bool()

    // Bereits gespeicherte H1-Highs auf Sweep prüfen.
    int k = array.size(histHighPrice) - 1

    while k >= 0
        float storedPrice = array.get(histHighPrice, k)

        bool broken = sweepByWick ?
             high >= storedPrice + syminfo.mintick :
             close >= storedPrice + syminfo.mintick

        if broken
            array.remove(histHighPrice, k)
            array.remove(histHighTime, k)
            array.remove(histHighSharp, k)

        k -= 1

    // Neuen bestätigten H1-Swing erkennen.
    [highSwing, highPrice, highTime, highSharp] = f_h1HighData()

    if highSwing and not na(highPrice) and not na(highTime)

        bool alreadyStored = false

        if array.size(histHighTime) > 0
            for n = 0 to array.size(histHighTime) - 1
                if array.get(histHighTime, n) == highTime
                    alreadyStored := true
                    break

        if not alreadyStored
            array.push(histHighPrice, highPrice)
            array.push(histHighTime, highTime)
            array.push(histHighSharp, highSharp)

    // Die drei nächsten noch offenen Highs bestimmen.
    float p1 = na
    float p2 = na
    float p3 = na

    int t1 = na
    int t2 = na
    int t3 = na

    bool s1 = false
    bool s2 = false
    bool s3 = false

    float d1 = na
    float d2 = na
    float d3 = na

    if array.size(histHighPrice) > 0
        for n = 0 to array.size(histHighPrice) - 1

            float p = array.get(histHighPrice, n)

            if p > close
                float d = p - close
                int t = array.get(histHighTime, n)
                bool s = array.get(histHighSharp, n)

                if na(d1) or d < d1
                    d3 := d2
                    p3 := p2
                    t3 := t2
                    s3 := s2

                    d2 := d1
                    p2 := p1
                    t2 := t1
                    s2 := s1

                    d1 := d
                    p1 := p
                    t1 := t
                    s1 := s

                else if na(d2) or d < d2
                    d3 := d2
                    p3 := p2
                    t3 := t2
                    s3 := s2

                    d2 := d
                    p2 := p
                    t2 := t
                    s2 := s

                else if na(d3) or d < d3
                    d3 := d
                    p3 := p
                    t3 := t
                    s3 := s

    [p1, t1, s1, p2, t2, s2, p3, t3, s3]


//──────────────────────────────────────────────────────────────────────
// KANONISCHER 3M-MIRROR: NUR DER LETZTE GERISSENE LEVEL
//
// Diese Logik greift NICHT in die vorhandene Level-Erzeugung ein.
// Diese Logik läuft immer im 3m-Kontext. H1-Swings werden darin explizit
// aus bestätigten 60m-Daten geholt. Dadurch bleiben Risszeit und Ergebnis
// nach 3m/H1-Timeframe-Wechseln identisch.
//──────────────────────────────────────────────────────────────────────

f_mirrorPriority(int code) =>
    int result = 1

    if code >= 30
        result := 5
    else if code >= 20
        result := 4
    else if code >= 10
        result := 3
    else if code == 2
        result := 2

    result


f_mirrorAddOrUpgrade(
    array<float> prices,
    array<bool> highs,
    array<int> codes,
    array<int> starts,
    array<int> created,
    float price,
    bool isHigh,
    int code,
    int startTime,
    int createdTime) =>

    int existing = -1

    if not na(price) and not na(startTime)
        if array.size(prices) > 0
            for i = 0 to array.size(prices) - 1
                if array.get(highs, i) == isHigh and f_samePrice(array.get(prices, i), price)
                    existing := i
                    break

        if existing >= 0
            int oldCode = array.get(codes, existing)

            if f_mirrorPriority(code) > f_mirrorPriority(oldCode)
                array.set(codes, existing, code)
                array.set(starts, existing, startTime)
        else
            array.push(prices, price)
            array.push(highs, isHigh)
            array.push(codes, code)
            array.push(starts, startTime)
            array.push(created, createdTime)

    true


f_h1LastSweptMirror() =>
    // Nur aktive, swing-basierte Levels werden hier gespeichert.
    var array<float> ePrice = array.new_float()
    var array<bool> eHigh = array.new_bool()
    var array<int> eCode = array.new_int()
    var array<int> eStart = array.new_int()
    var array<int> eCreated = array.new_int()

    var float lastPrice = na
    var int lastCode = 0
    var int lastStart = na
    var int lastSweep = na
    var bool lastIsHigh = false

    // London
    var float s1High = na
    var float s1Low = na
    var int s1HighTime = na
    var int s1LowTime = na
    var bool s1HighIsSwing = false
    var bool s1LowIsSwing = false
    var int s1HighSwingTime = na
    var int s1LowSwingTime = na
    var float pendingS1High = na
    var float pendingS1Low = na
    var int pendingS1HighStart = na
    var int pendingS1LowStart = na
    var int pendingS1HighSweep = na
    var int pendingS1LowSweep = na
    var int pendingS1HighCreated = na
    var int pendingS1LowCreated = na

    // NY AM
    var float s2High = na
    var float s2Low = na
    var bool s2HighIsSwing = false
    var bool s2LowIsSwing = false
    var int s2HighSwingTime = na
    var int s2LowSwingTime = na

    // NY PM
    var float s3High = na
    var float s3Low = na
    var bool s3HighIsSwing = false
    var bool s3LowIsSwing = false
    var int s3HighSwingTime = na
    var int s3LowSwingTime = na
    var float pendingS3High = na
    var float pendingS3Low = na

    // Day
    var float eDayHigh = na
    var float eDayLow = na
    var bool eDayHighIsSwing = false
    var bool eDayLowIsSwing = false
    var int eDayHighSwingTime = na
    var int eDayLowSwingTime = na
    var float ePendingDayHigh = na
    var float ePendingDayLow = na

    // Week
    var float eWeekHigh = na
    var float eWeekLow = na
    var bool eWeekHighIsSwing = false
    var bool eWeekLowIsSwing = false
    var int eWeekHighSwingTime = na
    var int eWeekLowSwingTime = na
    var float ePendingWeekHigh = na
    var float ePendingWeekLow = na

    
    //──────────────── London ────────────────
    bool eInS1 = not na(time(timeframe.period, session1, sessionTZ))

    if eInS1
        if not eInS1[1]
            s1High := high
            s1Low := low
            s1HighTime := time
            s1LowTime := time
            s1HighIsSwing := false
            s1LowIsSwing := false
            s1HighSwingTime := na
            s1LowSwingTime := na
        else
            if high > s1High
                s1High := high
                s1HighTime := time
                s1HighIsSwing := false
                s1HighSwingTime := na

            if low < s1Low
                s1Low := low
                s1LowTime := time
                s1LowIsSwing := false
                s1LowSwingTime := na

    if showSessions and showSession1 and not eInS1 and eInS1[1]
        if s1HighIsSwing
            f_mirrorAddOrUpgrade(ePrice, eHigh, eCode, eStart, eCreated, s1High, true, 11, s1HighSwingTime, time)
            pendingS1High := na
        else
            pendingS1High := s1High
            pendingS1HighStart := s1HighTime
            pendingS1HighSweep := na
            pendingS1HighCreated := time

        if s1LowIsSwing
            f_mirrorAddOrUpgrade(ePrice, eHigh, eCode, eStart, eCreated, s1Low, false, 11, s1LowSwingTime, time)
            pendingS1Low := na
        else
            pendingS1Low := s1Low
            pendingS1LowStart := s1LowTime
            pendingS1LowSweep := na
            pendingS1LowCreated := time

    // London-Level während Pending-Phase auf Sweep prüfen
    if not na(pendingS1High) and na(pendingS1HighSweep)
        bool pendingHighBroken = sweepByWick ? high >= pendingS1High + syminfo.mintick : close >= pendingS1High + syminfo.mintick

        if pendingHighBroken
            pendingS1HighSweep := time

    if not na(pendingS1Low) and na(pendingS1LowSweep)
        bool pendingLowBroken = sweepByWick ? low <= pendingS1Low - syminfo.mintick : close <= pendingS1Low - syminfo.mintick

        if pendingLowBroken
            pendingS1LowSweep := time

    //──────────────── NY AM ────────────────
    bool eInS2 = not na(time(timeframe.period, session2, sessionTZ))

    if eInS2
        if not eInS2[1]
            s2High := high
            s2Low := low
            s2HighIsSwing := false
            s2LowIsSwing := false
            s2HighSwingTime := na
            s2LowSwingTime := na
        else
            if high > s2High
                s2High := high
                s2HighIsSwing := false
                s2HighSwingTime := na

            if low < s2Low
                s2Low := low
                s2LowIsSwing := false
                s2LowSwingTime := na

    if showSessions and showSession2 and not eInS2 and eInS2[1]
        if s2HighIsSwing
            f_mirrorAddOrUpgrade(ePrice, eHigh, eCode, eStart, eCreated, s2High, true, 12, s2HighSwingTime, time)

        if s2LowIsSwing
            f_mirrorAddOrUpgrade(ePrice, eHigh, eCode, eStart, eCreated, s2Low, false, 12, s2LowSwingTime, time)

    //──────────────── NY PM ────────────────
    bool eInS3 = not na(time(timeframe.period, session3, sessionTZ))

    if eInS3
        if not eInS3[1]
            s3High := high
            s3Low := low
            s3HighIsSwing := false
            s3LowIsSwing := false
            s3HighSwingTime := na
            s3LowSwingTime := na
        else
            if high > s3High
                s3High := high
                s3HighIsSwing := false
                s3HighSwingTime := na

            if low < s3Low
                s3Low := low
                s3LowIsSwing := false
                s3LowSwingTime := na

    if showSessions and showSession3 and not eInS3 and eInS3[1]
        if s3HighIsSwing
            f_mirrorAddOrUpgrade(ePrice, eHigh, eCode, eStart, eCreated, s3High, true, 13, s3HighSwingTime, time)
            pendingS3High := na
        else
            pendingS3High := s3High

        if s3LowIsSwing
            f_mirrorAddOrUpgrade(ePrice, eHigh, eCode, eStart, eCreated, s3Low, false, 13, s3LowSwingTime, time)
            pendingS3Low := na
        else
            pendingS3Low := s3Low

    //──────────────── Day ────────────────
    int eNyYear = year(time, "America/New_York")
    int eNyMonth = month(time, "America/New_York")
    int eNyDay = dayofmonth(time, "America/New_York")
    bool eNewDay = eNyDay != eNyDay[1] or eNyMonth != eNyMonth[1] or eNyYear != eNyYear[1]

    if na(eDayHigh)
        eDayHigh := high
        eDayLow := low
    else if eNewDay
        if showPD
            if eDayHighIsSwing
                f_mirrorAddOrUpgrade(ePrice, eHigh, eCode, eStart, eCreated, eDayHigh, true, 20, eDayHighSwingTime, time)
                ePendingDayHigh := na
            else
                ePendingDayHigh := eDayHigh

            if eDayLowIsSwing
                f_mirrorAddOrUpgrade(ePrice, eHigh, eCode, eStart, eCreated, eDayLow, false, 20, eDayLowSwingTime, time)
                ePendingDayLow := na
            else
                ePendingDayLow := eDayLow

        eDayHigh := high
        eDayLow := low
        eDayHighIsSwing := false
        eDayLowIsSwing := false
        eDayHighSwingTime := na
        eDayLowSwingTime := na
    else
        if high > eDayHigh
            eDayHigh := high
            eDayHighIsSwing := false
            eDayHighSwingTime := na

        if low < eDayLow
            eDayLow := low
            eDayLowIsSwing := false
            eDayLowSwingTime := na

    //──────────────── Week ────────────────
    bool eNewWeek = timeframe.change("W")

    if na(eWeekHigh)
        eWeekHigh := high
        eWeekLow := low
    else if eNewWeek
        if showPW
            if eWeekHighIsSwing
                f_mirrorAddOrUpgrade(ePrice, eHigh, eCode, eStart, eCreated, eWeekHigh, true, 30, eWeekHighSwingTime, time)
                ePendingWeekHigh := na
            else
                ePendingWeekHigh := eWeekHigh

            if eWeekLowIsSwing
                f_mirrorAddOrUpgrade(ePrice, eHigh, eCode, eStart, eCreated, eWeekLow, false, 30, eWeekLowSwingTime, time)
                ePendingWeekLow := na
            else
                ePendingWeekLow := eWeekLow

        eWeekHigh := high
        eWeekLow := low
        eWeekHighIsSwing := false
        eWeekLowIsSwing := false
        eWeekHighSwingTime := na
        eWeekLowSwingTime := na
    else
        if high > eWeekHigh
            eWeekHigh := high
            eWeekHighIsSwing := false
            eWeekHighSwingTime := na

        if low < eWeekLow
            eWeekLow := low
            eWeekLowIsSwing := false
            eWeekLowSwingTime := na

    //──────────────── Neue H1-Swings ────────────────
    // Dieser Mirror läuft absichtlich im 3m-Kontext. Die H1-Swings werden
    // deshalb hier explizit aus dem bestätigten 60m-Kontext geholt.
    [eSwingHigh, eSwingHighPrice, eSwingHighTime, eSharpHigh] = request.security(
        syminfo.tickerid,
        "60",
        f_h1HighData(),
        lookahead=barmerge.lookahead_off)

    [eSwingLow, eSwingLowPrice, eSwingLowTime, eSharpLow] = request.security(
        syminfo.tickerid,
        "60",
        f_h1LowData(),
        lookahead=barmerge.lookahead_off)

    bool eNewSwingHigh = not na(eSwingHighTime) and (na(eSwingHighTime[1]) or eSwingHighTime != eSwingHighTime[1])
    bool eNewSwingLow = not na(eSwingLowTime) and (na(eSwingLowTime[1]) or eSwingLowTime != eSwingLowTime[1])

    if eNewSwingHigh
        // Falls ein bereits beendetes London-H vor seiner H1-Bestätigung
        // auf 3m gerissen wurde, darf derselbe Preis nicht als neuer Swing H
        // wieder auferstehen. Der gespeicherte London-Riss gewinnt.
        bool sweptPendingLondonHigh = showSessions and showSession1 and not na(pendingS1HighSweep) and f_samePrice(eSwingHighPrice, pendingS1High)

        if showSwings and not sweptPendingLondonHigh
            int highCode = showSharp and eSharpHigh ? 2 : 1
            f_mirrorAddOrUpgrade(ePrice, eHigh, eCode, eStart, eCreated, eSwingHighPrice, true, highCode, eSwingHighTime, time)

        if eInS1 and f_samePrice(eSwingHighPrice, s1High)
            s1HighIsSwing := true
            s1HighSwingTime := eSwingHighTime

        if showSessions and showSession1 and f_samePrice(eSwingHighPrice, pendingS1High)

            if not na(pendingS1HighSweep)

                bool pendingHighBecomesLatest = na(lastSweep) or pendingS1HighSweep > lastSweep or (pendingS1HighSweep == lastSweep and (not lastIsHigh or pendingS1High > lastPrice))

                if pendingHighBecomesLatest
                    lastPrice := pendingS1High
                    lastCode := 11
                    lastStart := pendingS1HighStart
                    lastSweep := pendingS1HighSweep
                    lastIsHigh := true

            else
                f_mirrorAddOrUpgrade(
                    ePrice,
                    eHigh,
                    eCode,
                    eStart,
                    eCreated,
                    pendingS1High,
                    true,
                    11,
                    pendingS1HighStart,
                    time)

            pendingS1High := na
            pendingS1HighStart := na
            pendingS1HighSweep := na
            pendingS1HighCreated := na

        if eInS2 and f_samePrice(eSwingHighPrice, s2High)
            s2HighIsSwing := true
            s2HighSwingTime := eSwingHighTime

        if eInS3 and f_samePrice(eSwingHighPrice, s3High)
            s3HighIsSwing := true
            s3HighSwingTime := eSwingHighTime

        if showSessions and showSession3 and f_samePrice(eSwingHighPrice, pendingS3High)
            f_mirrorAddOrUpgrade(ePrice, eHigh, eCode, eStart, eCreated, pendingS3High, true, 13, eSwingHighTime, time)
            pendingS3High := na

        if f_samePrice(eSwingHighPrice, eDayHigh)
            eDayHighIsSwing := true
            eDayHighSwingTime := eSwingHighTime

        if showPD and f_samePrice(eSwingHighPrice, ePendingDayHigh)
            f_mirrorAddOrUpgrade(ePrice, eHigh, eCode, eStart, eCreated, ePendingDayHigh, true, 20, eSwingHighTime, time)
            ePendingDayHigh := na

        if f_samePrice(eSwingHighPrice, eWeekHigh)
            eWeekHighIsSwing := true
            eWeekHighSwingTime := eSwingHighTime

        if showPW and f_samePrice(eSwingHighPrice, ePendingWeekHigh)
            f_mirrorAddOrUpgrade(ePrice, eHigh, eCode, eStart, eCreated, ePendingWeekHigh, true, 30, eSwingHighTime, time)
            ePendingWeekHigh := na

    if eNewSwingLow
        // Gleiches Prinzip für London-L: ein bereits auf 3m gerissenes
        // Pending-Level darf nach der H1-Bestätigung nicht als Swing L
        // wieder aktiv werden.
        bool sweptPendingLondonLow = showSessions and showSession1 and not na(pendingS1LowSweep) and f_samePrice(eSwingLowPrice, pendingS1Low)

        if showSwings and not sweptPendingLondonLow
            int lowCode = showSharp and eSharpLow ? 2 : 1
            f_mirrorAddOrUpgrade(ePrice, eHigh, eCode, eStart, eCreated, eSwingLowPrice, false, lowCode, eSwingLowTime, time)

        if eInS1 and f_samePrice(eSwingLowPrice, s1Low)
            s1LowIsSwing := true
            s1LowSwingTime := eSwingLowTime

        if showSessions and showSession1 and f_samePrice(eSwingLowPrice, pendingS1Low)

            if not na(pendingS1LowSweep)

                bool pendingLowBecomesLatest = na(lastSweep) or pendingS1LowSweep > lastSweep or (pendingS1LowSweep == lastSweep and (lastIsHigh or pendingS1Low < lastPrice))

                if pendingLowBecomesLatest
                    lastPrice := pendingS1Low
                    lastCode := 11
                    lastStart := pendingS1LowStart
                    lastSweep := pendingS1LowSweep
                    lastIsHigh := false

            else
                f_mirrorAddOrUpgrade(ePrice, eHigh, eCode, eStart, eCreated, pendingS1Low, false, 11, pendingS1LowStart, time)

            pendingS1Low := na
            pendingS1LowStart := na
            pendingS1LowSweep := na
            pendingS1LowCreated := na

        if eInS2 and f_samePrice(eSwingLowPrice, s2Low)
            s2LowIsSwing := true
            s2LowSwingTime := eSwingLowTime

        if eInS3 and f_samePrice(eSwingLowPrice, s3Low)
            s3LowIsSwing := true
            s3LowSwingTime := eSwingLowTime

        if showSessions and showSession3 and f_samePrice(eSwingLowPrice, pendingS3Low)
            f_mirrorAddOrUpgrade(ePrice, eHigh, eCode, eStart, eCreated, pendingS3Low, false, 13, eSwingLowTime, time)
            pendingS3Low := na

        if f_samePrice(eSwingLowPrice, eDayLow)
            eDayLowIsSwing := true
            eDayLowSwingTime := eSwingLowTime

        if showPD and f_samePrice(eSwingLowPrice, ePendingDayLow)
            f_mirrorAddOrUpgrade(ePrice, eHigh, eCode, eStart, eCreated, ePendingDayLow, false, 20, eSwingLowTime, time)
            ePendingDayLow := na

        if f_samePrice(eSwingLowPrice, eWeekLow)
            eWeekLowIsSwing := true
            eWeekLowSwingTime := eSwingLowTime

        if showPW and f_samePrice(eSwingLowPrice, ePendingWeekLow)
            f_mirrorAddOrUpgrade(ePrice, eHigh, eCode, eStart, eCreated, ePendingWeekLow, false, 30, eSwingLowTime, time)
            ePendingWeekLow := na

    //──────────────── Sweeps ────────────────
    int sweepIndex = 0

    while sweepIndex < array.size(ePrice)
        float p = array.get(ePrice, sweepIndex)
        bool isHigh = array.get(eHigh, sweepIndex)
        int createdTime = array.get(eCreated, sweepIndex)
        bool swept = false

        if time > createdTime
            if sweepByWick
                if isHigh
                    swept := high >= p + syminfo.mintick
                else
                    swept := low <= p - syminfo.mintick
            else
                if isHigh
                    swept := close >= p + syminfo.mintick
                else
                    swept := close <= p - syminfo.mintick

        if swept
            bool becomesLatest =
                 na(lastSweep) or
                 time > lastSweep or
                 (
                     time == lastSweep and
                     isHigh == lastIsHigh and
                     (
                         (isHigh and p > lastPrice) or
                         (not isHigh and p < lastPrice)
                     )
                 )

            if becomesLatest
                lastPrice := p
                lastCode := array.get(eCode, sweepIndex)
                lastStart := array.get(eStart, sweepIndex)
                lastSweep := time
                lastIsHigh := isHigh

            array.remove(ePrice, sweepIndex)
            array.remove(eHigh, sweepIndex)
            array.remove(eCode, sweepIndex)
            array.remove(eStart, sweepIndex)
            array.remove(eCreated, sweepIndex)
        else
            sweepIndex += 1

    [lastPrice, lastCode, lastStart, lastSweep, lastIsHigh]

    else if code == 11

    else if code == 12

    else if code == 13
        txt := isHigh ? "NY PM H" : "NY PM L"
    else if code == 20
        txt := isHigh ? "PDH" : "PDL"
    else if code == 30
        txt := isHigh ? "PWH" : "PWL"

    txt

       f_samePrice(
           h1SwingLowPrice,
           pendingDayLow)

        f_addOrUpgradeLevel(
             pendingDayLow,
             false,
             "PDL",
             colDay,
             4,
             h1SwingLowTime)

        pendingDayLow := na


    // aktuelle Woche
    if f_samePrice(
         h1SwingLowPrice,
         weekLow)

        weekLowIsSwing := true
        weekLowSwingTime := h1SwingLowTime


    // Vorwoche
    if showPW and
       f_samePrice(
           h1SwingLowPrice,
           pendingWeekLow)

        f_addOrUpgradeLevel(
             pendingWeekLow,
             false,
             "PWL",
             colWeek,
             5,
             h1SwingLowTime)

        pendingWeekLow := na

array<float> nearestLowDist = array.new_float()

        if array.get(levelActive, j)

            float candidatePrice = array.get(levelPrice, j)

            bool candidateIsHigh = array.get(levelHigh, j)

            if candidateIsHigh and candidatePrice > close

                float distance = candidatePrice - close

                f_keepNearest(
                    nearestHighIdx,
                    nearestHighDist,
                    j,
                    distance,
                    visiblePerSide)

            if not candidateIsHigh and candidatePrice < close

                float distance = close - candidatePrice

                f_keepNearest(
                    nearestLowIdx,
                    nearestLowDist,
                    j,
                    distance,
                    visiblePerSide)

            else if array.get(levelPriority, i) == 2

                originalColor := colSession
            else if array.get(levelPriority, i) == 4
                originalColor := colDay

                label.set_textcolor(lbl, originalColor)

                    if isHigh
                        swept := high >= price + syminfo.mintick
                    else
                        swept := low <= price - syminfo.mintick
                else
                    if isHigh
                        swept := close >= price + syminfo.mintick
                    else
                        swept := close <= price - syminfo.mintick
