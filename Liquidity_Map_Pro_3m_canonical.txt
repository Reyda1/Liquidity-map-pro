f_samePrice(float a, float b) =>
    not na(a) and
     not na(b) and
     math.abs(a - b) <= syminfo.mintick * 0.5

f_keepNearest(
    array<int> idxArr,
    array<float> distArr,
    int idx,
    float dist,
    int limit) =>

    int insertAt = array.size(distArr)

    if array.size(distArr) > 0
        for k = 0 to array.size(distArr) - 1
            if dist < array.get(distArr, k)
                insertAt := k
                break

    array.insert(idxArr, insertAt, idx)
    array.insert(distArr, insertAt, dist)

    if array.size(idxArr) > limit
        array.pop(idxArr)
        array.pop(distArr)

    true

float sharpDistance =
     sharpTicks * syminfo.mintick


// HIGH:
// Variante 1:
// links tiefer → Swing → rechts tiefer
//
// Variante 2:
// links tiefer → 2 gleiche Highs → rechts tiefer

f_h1HighData() =>

    // Nur abgeschlossene H1-Kerzen verwenden.
    // [1] = vollständig geschlossene rechte Bestätigungskerze
    // [2] = Swing-Kerze bei einfachem Swing

    bool singleHigh =
         high[2] > high[3] and
         high[2] > high[1]

    // Doppel-Swing:
    // links tiefer → zwei gleiche Highs → rechts tiefer
    bool doubleHigh =
         high[3] > high[4] and
         high[3] == high[2] and
         high[2] > high[1]

    bool isSwing =
         singleHigh or doubleHigh

    float swingPrice =
         isSwing ? high[2] : na

    int swingTime =
         doubleHigh ?
         time[3] :
         singleHigh ?
         time[2] :
         na

    float comparisonHigh =
         doubleHigh ?
         math.max(high[4], high[1]) :
         math.max(high[3], high[1])

    bool isSharp =
         isSwing and
         swingPrice - comparisonHigh >= sharpDistance

    [
        isSwing,
        swingPrice,
        swingTime,
        isSharp
    ]


// LOW:
// Variante 1:
// links höher → Swing → rechts höher
//
// Variante 2:
// links höher → 2 gleiche Lows → rechts höher

f_h1LowData() =>

    // Nur abgeschlossene H1-Kerzen verwenden.

    bool singleLow =
         low[2] < low[3] and
         low[2] < low[1]

    // Doppel-Swing:
    // links höher → zwei gleiche Lows → rechts höher
    bool doubleLow =
         low[3] < low[4] and
         low[3] == low[2] and
         low[2] < low[1]

    bool isSwing =
         singleLow or doubleLow

    float swingPrice =
         isSwing ? low[2] : na

    int swingTime =
         doubleLow ?
         time[3] :
         singleLow ?
         time[2] :
         na

    float comparisonLow =
         doubleLow ?
         math.min(low[4], low[1]) :
         math.min(low[3], low[1])

    bool isSharp =
         isSwing and
         comparisonLow - swingPrice >= sharpDistance

    [
        isSwing,
        swingPrice,
        swingTime,
        isSharp
    ]

f_mirrorSourceText(int code, bool isHigh) =>
    string txt = ""

    if code == 1
        txt := isHigh ? "Swing H" : "Swing L"
    else if code == 2
        txt := isHigh ? "SPITZ H" : "SPITZ L"
    else if code == 10
        txt := isHigh ? "Asia H" : "Asia L"
    else if code == 11
        txt := isHigh ? "London H" : "London L"
    else if code == 12
        txt := isHigh ? "NY AM H" : "NY AM L"
    else if code == 13
        txt := isHigh ? "NY PM H" : "NY PM L"
    else if code == 20
        txt := isHigh ? "PDH" : "PDL"
    else if code == 30
        txt := isHigh ? "PWH" : "PWL"

    txt


f_mirrorColor(int code) =>
    color c = colSwing

    if code >= 30
        c := colWeek
    else if code >= 20
        c := colDay
    else if code >= 10
        c := colSession
    else if code == 2
        c := colSharp

    c

f_mirrorPriority(int code) =>
    code >= 30 ? 5 : code >= 20 ? 4 : code >= 10 ? 3 : code == 2 ? 2 : 1

// Reine Datenabfrage, KEINE var-Arrays in request-Kontexten.
// Alle H1-Ausdrücke benutzen mindestens [1]. lookahead_on liefert diese
// bereits bestätigten Werte ab dem Anfang der nächsten H1, auch historisch.
f_confirmedH1() =>
    [sh, ph, th, sharpH] = f_h1HighData()
    [sl, pl, tl, sharpL] = f_h1LowData()
    [sh, ph, th, sharpH, sl, pl, tl, sharpL, time[2], time[3]]

type CanonicalBar
    int t
    int endTime
    float h
    float l
    float c
    int hourStart
    int dayKey
    int weekKey
    bool s0
    bool s1
    bool s2
    bool s3
    int s0Key
    int s1Key
    int s2Key
    int s3Key
    float swingH
    int swingHStart
    bool sharpH
    float swingL
    int swingLStart
    bool sharpL
    int swingLastOrigin
    int oldestSwingOrigin

f_canonicalBar() =>
    [sh, ph, th, sharpH, sl, pl, tl, sharpL, swingLastOrigin, oldestSwingOrigin] = request.security(
        syminfo.tickerid, "60", f_confirmedH1(),
        lookahead=barmerge.lookahead_on)
    int dayKey = year(time, "America/New_York") * 10000 + month(time, "America/New_York") * 100 + dayofmonth(time, "America/New_York")
    CanonicalBar.new(
        time, time_close, high, low, close, time("60"),
        dayKey, time("W"),
        not na(time("3", session0, sessionTZ)),
        not na(time("3", session1, sessionTZ)),
        not na(time("3", session2, sessionTZ)),
        not na(time("3", session3, sessionTZ)),
        time("1D", session0, sessionTZ), time("1D", session1, sessionTZ),
        time("1D", session2, sessionTZ), time("1D", session3, sessionTZ),
        sh ? ph : na, th, sharpH, sl ? pl : na, tl, sharpL, swingLastOrigin, oldestSwingOrigin)

type LiquidityLevel
    int id
    float price
    bool isHigh
    int code
    int start
    int created
    int sweptAt
    bool confirmedSwing
    // Preis allein ist keine Identität. Mehrere aktive Quellen dürfen
    // zusammenfallen; ihre H1-Ursprünge bleiben für spätere Upgrades erhalten.
    array<int> origins

type LiquidityPeriod
    bool running = false
    bool complete = false
    int key = na
    float hi = na
    float lo = na
    int hiTime = na
    int loTime = na
    int hiHour = na
    int loHour = na

type LiquidityState
    array<LiquidityLevel> levels
    array<LiquidityPeriod> periods
    int nextId = 0
    bool mergeNeeded = false
    int processed = na
    int highSignal = na
    int lowSignal = na
    LiquidityLevel lastSwept = na

f_hasOrigin(LiquidityLevel level, int origin, int lastOrigin) =>
    bool found = false
    for stamp in level.origins
        if stamp >= origin and stamp <= lastOrigin
            found := true
            break
    found

// Einziger Schreibpfad für Swing-, Session-, Tages- und Wochen-Level.
// sweptAt wird hier NIE zurückgesetzt, auch nicht bei einer H1-Bestätigung.
f_addOrUpgradeLevel(
    LiquidityState state, float price, bool isHigh, int code, int startTime,
    int origin, int lastOrigin, int created, bool confirmedSwing) =>
    int existing = -1
    int activeAtPrice = -1
    if not na(price) and not na(startTime) and not na(origin)
        state.mergeNeeded := true
        if array.size(state.levels) > 0
            for i = 0 to array.size(state.levels) - 1
                LiquidityLevel level = array.get(state.levels, i)
                if level.isHigh == isHigh and f_samePrice(level.price, price)
                    if f_hasOrigin(level, origin, lastOrigin)
                        existing := i
                        break
                    if na(level.sweptAt) and level.confirmedSwing
                        activeAtPrice := i
        // Ein gerissener gleicher Preis mit ANDEREM Ursprung ist kein
        // Verbot für ein tatsächlich neues Level an einem späteren Tag.
        if existing < 0
            existing := activeAtPrice
        if existing >= 0
            LiquidityLevel level = array.get(state.levels, existing)
            level.confirmedSwing := level.confirmedSwing or confirmedSwing
            if not array.includes(level.origins, origin)
                array.push(level.origins, origin)
            // Bei Doppel-Swings gehört auch die zweite H1 zum Ursprung.
            if lastOrigin != origin and not array.includes(level.origins, lastOrigin)
                array.push(level.origins, lastOrigin)
            if f_mirrorPriority(code) > f_mirrorPriority(level.code)
                level.code := code
                level.start := startTime
        else
            array<int> origins = array.from(origin)
            if lastOrigin != origin
                array.push(origins, lastOrigin)
            array.push(state.levels, LiquidityLevel.new(
                state.nextId, price, isHigh, code, startTime, created, na, confirmedSwing, origins))
            state.nextId += 1
    true

f_priceKey(LiquidityLevel level) =>
    int(math.round(level.price / syminfo.mintick)) * 2 + (level.isHigh ? 1 : 0)

f_mergeOpenLevels(LiquidityState state) =>
    // Nur bei neuen Quellen/Upgrades zusammenführen, nicht quadratisch
    // für jeden Intrabar. Map-Werte sind Referenzen auf dieselben Levels.
    map<int, LiquidityLevel> openByPrice = map.new<int, LiquidityLevel>()
    for level in state.levels
        if na(level.sweptAt) and level.confirmedSwing
            int key = f_priceKey(level)
            if not map.contains(openByPrice, key)
                map.put(openByPrice, key, level)
    int i = 0
    while i < array.size(state.levels)
        LiquidityLevel other = array.get(state.levels, i)
        int key = f_priceKey(other)
        bool removed = false
        if na(other.sweptAt) and map.contains(openByPrice, key)
            LiquidityLevel target = map.get(openByPrice, key)
            if target.id != other.id
                target.created := math.min(target.created, other.created)
                for stamp in other.origins
                    if not array.includes(target.origins, stamp)
                        array.push(target.origins, stamp)
                if f_mirrorPriority(other.code) > f_mirrorPriority(target.code)
                    target.code := other.code
                    target.start := other.start
                array.remove(state.levels, i)
                removed := true
        if not removed
            i += 1
    state.mergeNeeded := false
    true

f_enabled(LiquidityLevel level) =>
    // PD und NY PM waren bereits über die früheren H1-Snapshots auch
    // ohne Swing sichtbar. Diese Quellen bleiben unabhängig davon gültig.
    (level.confirmedSwing or level.code == 13 or level.code == 20) and (level.code >= 10 or showSwings)

// Keine rekonstruierte Reihenfolge innerhalb eines 3m-OHLC-Bars erfinden.
// Gleichzeitige Risse: High vor Low, außen vor innen, dann Priorität und
// Ursprung. Damit hängt der Gewinner nicht von der Array-Reihenfolge ab.
f_laterSweep(LiquidityLevel candidate, LiquidityLevel previous) =>
    bool later = na(previous)
    if not later
        if candidate.sweptAt != previous.sweptAt
            later := candidate.sweptAt > previous.sweptAt
        else if candidate.isHigh != previous.isHigh
            later := candidate.isHigh
        else if not f_samePrice(candidate.price, previous.price)
            later := candidate.isHigh ? candidate.price > previous.price : candidate.price < previous.price
        else if f_mirrorPriority(candidate.code) != f_mirrorPriority(previous.code)
            later := f_mirrorPriority(candidate.code) > f_mirrorPriority(previous.code)
        else
            later := candidate.start > previous.start
    later

// Dieselbe Pending-Logik für ALLE Sessions, PD und PW. Auch noch nicht
// H1-bestätigte abgeschlossene Perioden nehmen bereits an Sweeps teil.
f_updatePeriod(LiquidityState state, int slot, bool inside, int key, bool enabled, int code, CanonicalBar b) =>
    LiquidityPeriod period = array.get(state.periods, slot)
    bool boundary = period.running and (not inside or key != period.key)
    if boundary and period.complete and enabled
        f_addOrUpgradeLevel(state, period.hi, true, code, period.hiTime,
            period.hiHour, period.hiHour, b.t, false)
        f_addOrUpgradeLevel(state, period.lo, false, code, period.loTime,
            period.loHour, period.loHour, b.t, false)
    if inside
        if not period.running or boundary
            // Eine beim Datenbeginn nur teilweise vorhandene Periode nicht
            // als vollständige Session / vollständigen Tag ausgeben.
            period.complete := not na(state.processed)
            period.key := key
            period.hi := b.h
            period.lo := b.l
            period.hiTime := b.t
            period.loTime := b.t
            period.hiHour := b.hourStart
            period.loHour := b.hourStart
        else
            if b.h > period.hi
                period.hi := b.h
                period.hiTime := b.t
                period.hiHour := b.hourStart
            if b.l < period.lo
                period.lo := b.l
                period.loTime := b.t
                period.loHour := b.hourStart
    period.running := inside
    true

f_processBar(LiquidityState state, CanonicalBar b) =>
    if na(state.processed) or b.t > state.processed
        // Erst H1-Bestätigungen am Bar-Anfang, dann Periodenabschluss,
        // dann Sweep durch DIESEN 3m-Bar. Kein ganzer H1-Wick als Ersatz.
        if not na(b.swingH) and (na(state.highSignal) or b.swingHStart != state.highSignal)
            f_addOrUpgradeLevel(state, b.swingH, true, showSharp and b.sharpH ? 2 : 1,
                b.swingHStart, b.swingHStart, b.swingLastOrigin, b.t, true)
            state.highSignal := b.swingHStart
        if not na(b.swingL) and (na(state.lowSignal) or b.swingLStart != state.lowSignal)
            f_addOrUpgradeLevel(state, b.swingL, false, showSharp and b.sharpL ? 2 : 1,
                b.swingLStart, b.swingLStart, b.swingLastOrigin, b.t, true)
            state.lowSignal := b.swingLStart

        f_updatePeriod(state, 0, b.s0, b.s0Key, showSessions and showSession0, 10, b)
        f_updatePeriod(state, 1, b.s1, b.s1Key, showSessions and showSession1, 11, b)
        f_updatePeriod(state, 2, b.s2, b.s2Key, showSessions and showSession2, 12, b)
        f_updatePeriod(state, 3, b.s3, b.s3Key, showSessions and showSession3, 13, b)
        f_updatePeriod(state, 4, true, b.dayKey, showPD, 20, b)
        f_updatePeriod(state, 5, true, b.weekKey, showPW, 30, b)
        if state.mergeNeeded
            f_mergeOpenLevels(state)

        if array.size(state.levels) > 0
            for level in state.levels
                // Levels entstehen am Bar-Anfang. Bereits der erste Bar
                // nach Sessionende / Bestätigung kann sie reißen.
                if na(level.sweptAt) and b.t >= level.created
                    bool broken = (level.isHigh ?
                        (sweepByWick ? b.h : b.c) >= level.price + syminfo.mintick :
                        (sweepByWick ? b.l : b.c) <= level.price - syminfo.mintick)
                    if broken
                        level.sweptAt := b.t
                if not na(level.sweptAt) and f_enabled(level)
                    if f_laterSweep(level, state.lastSwept)
                        state.lastSwept := level

        // maxLevels begrenzt alte Historie, niemals offene Liquidität.
        // Pending/Tombstones bis nach dem spätestmöglichen H1-Signal
        // behalten. Die H1-Barfolge statt Wandzeit benutzen: Handelslücken
        // und verkürzte H1-Kerzen dürfen keine Tombstones vorzeitig löschen.
        int i = 0
        while i < array.size(state.levels)
            LiquidityLevel level = array.get(state.levels, i)
            int newestOrigin = array.max(level.origins)
            bool expired = not na(b.oldestSwingOrigin) and newestOrigin < b.oldestSwingOrigin
            bool oldSwept = not na(level.sweptAt) and array.size(state.levels) > maxLevels
            bool unusedPending = not level.confirmedSwing and level.code != 13 and level.code != 20
            if expired and (oldSwept or unusedPending) and (na(state.lastSwept) or level.id != state.lastSwept.id)
                array.remove(state.levels, i)
            else
                i += 1
        state.processed := b.t
    true

var LiquidityState liquidity = LiquidityState.new(array.new<LiquidityLevel>(), array.new<LiquidityPeriod>())
if barstate.isfirst
    for i = 0 to 5
        array.push(liquidity.periods, LiquidityPeriod.new())

// request.security_lower_tf liefert ausschließlich Rohdaten/Bestätigungen.
// Die zustandsbehaftete Funktion wird NICHT in einem request ausgewertet.
// Auf H1 werden dieselben 3m-Bars explizit aufsteigend durchlaufen.
// Keine varip-Zustände: Realtime-Rollback setzt den Zustand auf den Beginn
// der Chartkerze zurück; alle verfügbaren geschlossenen Intrabars werden
// bei jedem Update erneut abgespielt. Das ist auch nach Reload identisch.
float chartSeconds = timeframe.in_seconds()
if na(chartSeconds) or chartSeconds < 180 or chartSeconds % 180 != 0
    runtime.error("Liquidity Map Pro benötigt einen zeitbasierten Chart ab 3 Minuten, dessen Intervall durch 3 Minuten teilbar ist.")

if chartSeconds == 180
    CanonicalBar currentBar = f_canonicalBar()
    if barstate.isconfirmed
        f_processBar(liquidity, currentBar)
else
    array<CanonicalBar> intrabars = request.security_lower_tf(
        syminfo.tickerid, "3", f_canonicalBar(), calc_bars_count=100000)
    for b in intrabars
        if b.endTime <= timenow
            f_processBar(liquidity, b)

var array<line> drawnLines = array.new<line>()
var array<label> drawnLabels = array.new<label>()

// Nur Zeichenkoordinaten: tatsächliche H1-Intervalle inklusive Session-Versatz.
var array<int> drawH1Opens = array.new_int()
var array<int> drawH1Closes = array.new_int()
if chartSeconds == 3600
    array.push(drawH1Opens, time)
    array.push(drawH1Closes, time_close)

f_h1DrawTime(int exactTime) =>
    int drawTime = exactTime
    if chartSeconds == 3600
        int left = 0
        int right = array.size(drawH1Opens) - 1
        int containing = -1
        while left <= right
            int middle = int(math.floor((left + right) / 2))
            if array.get(drawH1Opens, middle) <= exactTime
                containing := middle
                left := middle + 1
            else
                right := middle - 1
        if containing >= 0
            if exactTime < array.get(drawH1Closes, containing)
                drawTime := array.get(drawH1Opens, containing)
    drawTime

f_drawLevel(LiquidityLevel level, int asOf, bool latest) =>
    bool active = na(level.sweptAt)
    color base = f_mirrorColor(level.code)
    color tint = active or latest ? base : colSwept
    int endTime = active ? asOf : level.sweptAt
    int drawStart = level.start
    int drawEnd = f_h1DrawTime(endTime)
    if chartSeconds == 3600
        // origins enthält auch nach Upgrades die H1-Ursprünge aller Quellen.
        // Der zum exakten Start gehörende Ursprung kann vor der Charthistorie liegen.
        int originHour = na
        for origin in level.origins
            if origin <= level.start and (na(originHour) or origin > originHour)
                originHour := origin
        drawStart := na(originHour) ? f_h1DrawTime(level.start) : originHour
    line ln = line.new(drawStart, level.price, drawEnd, level.price,
        xloc=xloc.bar_time, extend=active ? extend.right : extend.none,
        color=tint, style=active or latest ? line.style_solid : line.style_dashed,
        width=f_mirrorPriority(level.code) >= 4 ? 2 : 1)
    int labelTime = active ? math.max(level.start, asOf - 120 * 60 * 1000) : drawStart + int(math.round((drawEnd - drawStart) * 0.7))
    label lbl = label.new(labelTime, level.price,
        f_mirrorSourceText(level.code, level.isHigh) + (active ? "" : " ✓"),
        xloc=xloc.bar_time, style=label.style_none,
        textcolor=tint, color=color.new(color.black, 100), size=labelSizeValue)
    array.push(drawnLines, ln)
    array.push(drawnLabels, lbl)
    true

if barstate.islast
    for ln in drawnLines
        line.delete(ln)
    for lbl in drawnLabels
        label.delete(lbl)
    array.clear(drawnLines)
    array.clear(drawnLabels)

    array<int> nearestHighIdx = array.new_int()
    array<float> nearestHighDist = array.new_float()
    array<int> nearestLowIdx = array.new_int()
    array<float> nearestLowDist = array.new_float()
    if array.size(liquidity.levels) > 0
        for i = 0 to array.size(liquidity.levels) - 1
            LiquidityLevel level = array.get(liquidity.levels, i)
            if f_enabled(level) and na(level.sweptAt)
                if level.isHigh and level.price > close
                    f_keepNearest(nearestHighIdx, nearestHighDist, i, level.price - close, visiblePerSide)
                if not level.isHigh and level.price < close
                    f_keepNearest(nearestLowIdx, nearestLowDist, i, close - level.price, visiblePerSide)

    if not na(liquidity.processed)
        for i in nearestHighIdx
            f_drawLevel(array.get(liquidity.levels, i), liquidity.processed, false)
        for i in nearestLowIdx
            f_drawLevel(array.get(liquidity.levels, i), liquidity.processed, false)
        // Letzten Riss separat zeichnen, damit History-Limit und showSwept
        // seine Sichtbarkeit nicht beeinflussen.
        if not na(liquidity.lastSwept)
            f_drawLevel(liquidity.lastSwept, liquidity.processed, true)
        int remaining = math.max(0, maxLevels - array.size(drawnLines))
        int i = array.size(liquidity.levels) - 1
        while showSwept and remaining > 0 and i >= 0
            LiquidityLevel level = array.get(liquidity.levels, i)
            if not na(level.sweptAt) and f_enabled(level) and (na(liquidity.lastSwept) or level.id != liquidity.lastSwept.id)
                f_drawLevel(level, liquidity.processed, false)
                remaining -= 1
            i -= 1
